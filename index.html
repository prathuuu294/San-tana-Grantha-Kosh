    async function fetchVerifiedHinduArtwork(subjectName, count = 5) {

      try {

        const cleanSubj = (subjectName || "").replace(/[^a-zA-Z\s]/g, "").trim();

        if (!cleanSubj) return [];

        const query = `filetype:bitmap "${cleanSubj}" (Ramayana OR Mahabharata OR Purana OR Hindu OR Krishna OR Shiva OR Vishnu OR Devi OR "Ravi Varma")`;

        const url = `https://commons.wikimedia.org/w/api.php?action=query&generator=search&gsrsearch=${encodeURIComponent(query)}&gsrnamespace=6&gsrlimit=24&prop=imageinfo&iiprop=url|size&iiurlwidth=900&format=json&origin=*`;

        const res = await fetch(url);

        const json = await res.json();

        const pages = Object.values(json?.query?.pages || {});

        const subjTokens = cleanSubj.toLowerCase().split(/\s+/).filter(w => w.length >= 4);

        const results = [];

        for (const p of pages) {

          const info = p?.imageinfo?.[0];

          const imgUrl = info?.thumburl || info?.url || "";

          const title = p?.title || "";

          const lower = title.toLowerCase();

          if (!imgUrl || lower.endsWith(".svg") || lower.endsWith(".pdf") || lower.endsWith(".tif")) continue;

          if (BLOCKED_TERMS.some(b => lower.includes(b))) continue;

          // Must contain the actual character/subject name in the file title

          const hasSubject = subjTokens.length === 0 || subjTokens.some(tok => lower.includes(tok));

          // Must also match Hindu mythological context

          const hasHinduContext = HINDU_CONTEXT_TERMS.some(ctx => lower.includes(ctx));

          if (!hasSubject || !hasHinduContext) continue;

          results.push({

            url: imgUrl,

            caption: title.replace(/^File:/i, "").replace(/\.[^.]+$/, "").replace(/_/g, " "),

            source: "Sacred Epic Painting",

            isAiMovie: false

          });

          if (results.length >= count) break;

        }

        return results;

      } catch (e) {

        return [];

      }

    }
